#### Day 5-7：实现Selector节点

**任务**：实现论文筛选Agent，基于相关性评分

```python
from sentence_transformers import SentenceTransformer, util

class SelectorAgent:
    """论文筛选Agent"""
    
    def __init__(self):
        self.model = SentenceTransformer('all-MiniLM-L6-v2')
    
    def compute_relevance(self, papers: List[Paper], query: str) -> List[Paper]:
        """计算相关性分数"""
        query_embedding = self.model.encode(query, convert_to_tensor=True)
        
        for paper in papers:
            text = f"{paper.title} {paper.abstract}"
            paper_embedding = self.model.encode(text, convert_to_tensor=True)
            
            similarity = util.cos_sim(query_embedding, paper_embedding).item()
            paper.relevance_score = similarity
        
        return papers
    
    def filter_by_criteria(self, papers: List[Paper], 
                          min_citations: int = 0,
                          min_year: int = 2015,
                          max_results: int = 30) -> List[Paper]:
        """按条件筛选"""
        filtered = [
            p for p in papers 
            if p.citations >= min_citations and p.year >= min_year
        ]
        
        # 按相关性排序
        filtered.sort(key=lambda p: p.relevance_score, reverse=True)
        
        return filtered[:max_results]
    
    def cross_encoder_rerank(self, query: str, papers: List[Paper], 
                            top_k: int = 20) -> List[Paper]:
        """Cross-Encoder重排序"""
        try:
            from sentence_transformers import CrossEncoder
            model = CrossEncoder('cross-encoder/ms-marco-MiniLM-L-6-v2')
            
            pairs = [(query, f"{p.title} {p.abstract}") for p in papers]
            scores = model.predict(pairs)
            
            for paper, score in zip(papers, scores):
                paper.relevance_score = score
            
            papers.sort(key=lambda p: p.relevance_score, reverse=True)
            
        except Exception as e:
            print(f"Cross-Encoder重排序失败: {e}，使用原有分数")
        
        return papers[:top_k]
    
    def execute(self, state: PaperAgentState) -> PaperAgentState:
        """执行筛选任务"""
        papers = state["papers"]
        query = state["user_query"]
        
        # 计算相关性
        papers = self.compute_relevance(papers, query)
        
        # 条件筛选
        papers = self.filter_by_criteria(
            papers, 
            min_citations=0,  # 新论文可能引用少
            min_year=2020,
            max_results=50
        )
        
        # Cross-Encoder重排序
        papers = self.cross_encoder_rerank(query, papers, top_k=20)
        
        return {"selected_papers": papers}
```

```python
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver

def build_paper_workflow() -> StateGraph:
    """构建论文工作流"""
    
    # 创建图
    workflow = StateGraph(PaperAgentState)
    
    # 添加节点
    workflow.add_node("crawler", crawler_agent.execute)
    workflow.add_node("selector", selector_agent.execute)
    workflow.add_node("outline", outline_agent.execute)
    workflow.add_node("writer", writer_agent.execute)
    workflow.add_node("reviewer", reviewer_agent.execute)
    
    # 定义边
    workflow.set_entry_point("crawler")
    workflow.add_edge("crawler", "selector")
    workflow.add_edge("selector", "outline")
    workflow.add_edge("outline", "writer")
    workflow.add_edge("writer", "reviewer")
    
    # 条件路由
    def should_continue(state: PaperAgentState) -> str:
        if len(state["feedback"]) == 0 or state["iteration"] >= state["max_iterations"]:
            return END
        return "writer"
    
    workflow.add_conditional_edges(
        "reviewer",
        should_continue,
        {END: END, "writer": "writer"}
    )
    
    # 编译
    memory = MemorySaver()
    app = workflow.compile(checkpointer=memory)
    
    return app

# 执行工作流
async def run_paper_workflow(query: str, user_id: str = "default"):
    """执行论文工作流"""
    app = build_paper_workflow()
    
    initial_state = PaperAgentState(
        user_query=query,
        papers=[],
        selected_papers=[],
        outline={},
        draft="",
        feedback=[],
        iteration=0,
        max_iterations=3,
        current_phase="start",
        session_id=str(uuid.uuid4()),
        user_id=user_id,
        timestamp=time.time()
    )
    
    result = await app.ainvoke(initial_state)
    
    return result
```

```python
from neo4j import GraphDatabase

class Neo4jMemory:
    """Neo4j图记忆"""
    
    def __init__(self, uri="bolt://localhost:7687", username="neo4j", password="password"):
        self.driver = GraphDatabase.driver(uri, auth=(username, password))
    
    def add_paper(self, paper: Paper):
        """添加论文到图谱"""
        with self.driver.session() as session:
            # 创建论文节点
            session.run("""
                MERGE (p:Paper {id: $id})
                SET p.title = $title,
                    p.year = $year,
                    p.abstract = $abstract,
                    p.citations = $citations
            """, id=paper.id, title=paper.title, year=paper.year, 
                abstract=paper.abstract, citations=paper.citations)
            
            # 创建作者节点并连接
            for author in paper.authors:
                session.run("""
                    MERGE (a:Author {name: $name})
                    MERGE (p:Paper {id: $paper_id})
                    MERGE (a)-[:AUTHORED]->(p)
                """, name=author, paper_id=paper.id)
    
    def add_citation(self, from_id: str, to_id: str):
        """添加引用关系"""
        with self.driver.session() as session:
            session.run("""
                MATCH (from:Paper {id: $from_id})
                MATCH (to:Paper {id: $to_id})
                MERGE (from)-[:CITES]->(to)
            """, from_id=from_id, to_id=to_id)
    
    def find_related_papers(self, paper_id: str, depth: int = 2) -> List[Paper]:
        """查找相关论文"""
        with self.driver.session() as session:
            result = session.run("""
                MATCH (p:Paper {id: $id})-[:CITES*1..$depth]->(related:Paper)
                RETURN related
                LIMIT 50
            """, id=paper_id, depth=depth)
            
            papers = []
            for record in result:
                paper_data = record["related"]
                papers.append(Paper(
                    id=paper_data["id"],
                    title=paper_data.get("title", ""),
                    authors=[],
                    abstract=paper_data.get("abstract", ""),
                    url="",
                    year=paper_data.get("year", 0),
                    citations=paper_data.get("citations", 0)
                ))
            
            return papers
    
    def close(self):
        self.driver.close()
```

```python
class HybridMemoryManager:
    """混合记忆管理器"""
    
    def __init__(self):
        # 向量记忆
        self.vector_store = ChromaDB()
        
        # 图记忆
        self.graph_store = Neo4jMemory()
        
        # 遗忘曲线
        self.forgetting_curve = ForgettingCurveMemory()
    
    async def store(self, paper: Paper, user_id: str):
        """多层存储"""
        # 向量存储
        await self.vector_store.add(
            content=f"{paper.title} {paper.abstract}",
            metadata={"paper_id": paper.id, "user_id": user_id}
        )
        
        # 图存储
        self.graph_store.add_paper(paper)
        
        # 遗忘曲线
        self.forgetting_curve.add(
            content=paper.title,
            importance=self._calculate_importance(paper)
        )
    
    async def retrieve(self, query: str, mode: str = "hybrid", 
                      top_k: int = 20) -> List[Paper]:
        """混合检索"""
        if mode == "vector":
            return await self.vector_store.search(query, top_k=top_k)
        elif mode == "graph":
            return self.graph_store.find_related_papers(query, depth=2)
        else:  # hybrid
            vector_results = await self.vector_store.search(query, top_k=20)
            # 融合结果
            return self._fuse_results(vector_results, top_k)
    
    def _calculate_importance(self, paper: Paper) -> float:
        """计算论文重要性"""
        return min(10.0, (paper.citations / 100.0) * 5 + 5.0)
    
    def _fuse_results(self, results: List, top_k: int) -> List:
        """融合检索结果"""
        # 简单融合：按相关性排序
        results.sort(key=lambda x: x.get("score", 0), reverse=True)
        return results[:top_k]
```

```python
        if profile.writing_style == "detailed":
            query += " comprehensive survey review"
        elif profile.writing_style == "concise":
            query += " brief overview summary"
        
        return query
    
    def _rank_by_preference(self, papers: List[Paper], profile) -> List[Paper]:
        """根据偏好排序"""
        # 根据用户兴趣调整分数
        for paper in papers:
            if any(topic in paper.title.lower() for topic in profile.topics):
                paper.relevance_score *= 1.2
        
        papers.sort(key=lambda p: p.relevance_score, reverse=True)
        return papers
```

```python
from transformers import CLIPProcessor, CLIPModel
from PIL import Image
import fitz  # PyMuPDF

class MultimodalPaperAgent:
    """多模态论文Agent"""
    
    def __init__(self):
        # CLIP模型
        self.clip_model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
        self.clip_processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")
    
    def extract_images_from_pdf(self, pdf_path: str) -> List[Image.Image]:
        """从PDF提取图像"""
        doc = fitz.open(pdf_path)
        images = []
        
        for page_num in range(len(doc)):
            page = doc[page_num]
            image_list = page.get_images()
            
            for img_index, img in enumerate(image_list):
                xref = img[0]
                base_image = doc.extract_image(xref)
                image_bytes = base_image["image"]
                
                image = Image.open(BytesIO(image_bytes))
                images.append(image)
        
        return images
    
    def encode_image(self, image: Image.Image) -> np.ndarray:
        """编码图像"""
        inputs = self.clip_processor(images=image, return_tensors="pt")
        return self.clip_model.get_image_features(**inputs).detach().numpy()
    
    def encode_text(self, text: str) -> np.ndarray:
        """编码文本"""
        inputs = self.clip_processor(text=text, return_tensors="pt")
        return self.clip_model.get_text_features(**inputs).detach().numpy()
    
    def find_similar_images(self, query: str, images: List[Image.Image], 
                           top_k: int = 5) -> List[Tuple[int, float]]:
        """查找相似图像"""
        query_embedding = self.encode_text(query)
        
        similarities = []
        for i, img in enumerate(images):
            img_embedding = self.encode_image(img)
            sim = cosine_similarity(query_embedding, img_embedding)
            similarities.append((i, sim))
        
        similarities.sort(key=lambda x: x[1], reverse=True)
        return similarities[:top_k]
```

```python
import spacy
from typing import List, Tuple

class KnowledgeGraphBuilder:
    """知识图谱构建器"""
    
    def __init__(self):
        self.nlp = spacy.load("en_core_web_sm")
    
    def extract_entities(self, text: str) -> List[Tuple[str, str]]:
        """提取实体"""
        doc = self.nlp(text)
        
        entities = []
        for ent in doc.ents:
            entities.append((ent.text, ent.label_))
        
        return entities
    
    def extract_relations(self, text: str) -> List[Tuple[str, str, str]]:
        """提取关系"""
        doc = self.nlp(text)
        
        relations = []
        for token in doc:
            if token.dep_ in ("nsubj", "dobj"):
                head = token.head
                relations.append((token.text, token.dep_, head.text))
        
        return relations
    
    def build_graph_from_papers(self, papers: List[Paper]) -> nx.DiGraph:
        """从论文构建知识图谱"""
        G = nx.DiGraph()
        
        for paper in papers:
            # 添加论文节点
            G.add_node(paper.id, type="paper", title=paper.title)
            
            # 提取实体
            entities = self.extract_entities(f"{paper.title} {paper.abstract}")
            
            for entity, entity_type in entities:
                G.add_node(entity, type=entity_type)
                G.add_edge(paper.id, entity, relation="mentions")
            
            # 添加引用关系
            for ref in getattr(paper, 'references', []):
                G.add_edge(paper.id, ref, relation="cites")
        
        return G
```

```python
class PaperAgentEvaluator:
    """论文Agent评估器"""
    
    def evaluate_retrieval(self, papers: List[Paper], 
                          ground_truth: List[str]) -> Dict:
        """评估检索质量"""
        paper_ids = {p.id for p in papers}
        
        # Precision@K
        precision = len(paper_ids & set(ground_truth)) / len(paper_ids)
        
        # Recall@K
        recall = len(paper_ids & set(ground_truth)) / len(ground_truth)
        
        # F1
        f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
        
        return {
            "precision@k": precision,
            "recall@k": recall,
            "f1@k": f1,
            "total_retrieved": len(papers)
        }
    
    def evaluate_writing_quality(self, draft: str) -> Dict:
        """评估写作质量"""

{draft}
```
