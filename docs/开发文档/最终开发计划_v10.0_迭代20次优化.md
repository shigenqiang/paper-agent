### 1. 多智能体编排架构（LangGraph）

```python
from langgraph.graph import StateGraph, END

class PaperAgentState(TypedDict):
    """统一状态管理"""
    query: str
    papers: List[Paper]
    outline: Dict
    draft: str
    feedback: List[str]
    iteration: int

# 定义Agent节点
def crawler_agent(state: PaperAgentState) -> PaperAgentState:
    """论文爬取Agent"""
    papers = search_papers(state["query"])
    return {"papers": papers}

def selector_agent(state: PaperAgentState) -> PaperAgentState:
    """论文筛选Agent"""
    relevant = filter_papers(state["papers"], state["query"])
    return {"papers": relevant}

def outline_agent(state: PaperAgentState) -> PaperAgentState:
    """大纲生成Agent"""
    outline = generate_outline(state["papers"])
    return {"outline": outline}

def writing_agent(state: PaperAgentState) -> PaperAgentState:
    """写作Agent"""
    draft = write_section(state["outline"], state["papers"])
    return {"draft": draft}

def review_agent(state: PaperAgentState) -> PaperAgentState:
    """审查Agent"""
    feedback = review_draft(state["draft"])
    return {"feedback": feedback}

# 构建工作流图
workflow = StateGraph(PaperAgentState)
workflow.add_node("crawler", crawler_agent)
workflow.add_node("selector", selector_agent)
workflow.add_node("outline", outline_agent)
workflow.add_node("writing", writing_agent)
workflow.add_node("review", review_agent)

# 定义边和条件路由
workflow.add_edge("crawler", "selector")
workflow.add_edge("selector", "outline")
workflow.add_edge("outline", "writing")
workflow.add_edge("writing", "review")

def should_continue(state: PaperAgentState) -> str:
    if len(state["feedback"]) == 0 or state["iteration"] >= 3:
        return END
    return "writing"

workflow.add_conditional_edges("review", should_continue)
workflow.set_entry_point("crawler")

app = workflow.compile()
```

### 2. 混合记忆架构（Vector + Graph + Episodic）

```python
class HybridMemorySystem:
    """混合记忆系统"""
    
    def __init__(self):
        # 向量记忆：语义检索
        self.vector_store = ChromaDB()
        
        # 图记忆：关系推理
        self.graph_store = Neo4j()
        
        # 情节记忆：时序事件
        self.episodic_store = EpisodicMemory()
        
        # 遗忘曲线管理
        self.forgetting_curve = ForgettingCurveMemory()
    
    async def remember(self, content: str, context: Dict):
        """多层记忆存储"""
        # 1. 向量存储（语义）
        embedding = await self.embed(content)
        await self.vector_store.add(embedding, content)
        
        # 2. 图存储（关系）
        entities = extract_entities(content)
        relations = extract_relations(content)
        await self.graph_store.add_nodes(entities)
        await self.graph_store.add_edges(relations)
        
        # 3. 情节存储（时序）
        episode = {
            "content": content,
            "timestamp": time.time(),
            "context": context
        }
        await self.episodic_store.add(episode)
        
        # 4. 遗忘曲线管理
        importance = calculate_importance(content, context)
        await self.forgetting_curve.add(content, importance)
    
    async def recall(self, query: str, mode: str = "hybrid"):
        """混合检索"""
        if mode == "semantic":
            return await self.vector_store.search(query)
        elif mode == "relational":
            return await self.graph_store.query(query)
        elif mode == "temporal":
            return await self.episodic_store.search(query)
        else:  # hybrid
            # 融合三种检索结果
            semantic_results = await self.vector_store.search(query, top_k=20)
            graph_results = await self.graph_store.query(query, depth=2)
            episodic_results = await self.episodic_store.search(query, window=30)
            
            # 重排序和融合
            return self.fuse_results(semantic_results, graph_results, episodic_results)
```

### 3. 双智能体检索架构（参考PaSa）

```python
class DualAgentRetrieval:
    """双智能体检索系统"""
    
    def __init__(self):
        self.crawler = CrawlerAgent()
        self.selector = SelectorAgent()
        self.reranker = CrossEncoderReranker()
    
    async def search(self, query: str, max_iterations: int = 3):
        """迭代检索"""
        all_papers = []
        visited = set()
        
        for iteration in range(max_iterations):
            # 1. Crawler: 搜索和扩展
            new_papers = await self.crawler.search(query)
            
            # 2. 引用网络扩展
            for paper in new_papers:
                if paper.id not in visited:
                    visited.add(paper.id)
                    all_papers.append(paper)
                    
                    # 爬取引用
                    citations = await self.crawler.get_citations(paper.id)
                    all_papers.extend(citations)
            
            # 3. Selector: 相关性判断
            relevant_papers = await self.selector.select(all_papers, query)
            
            # 4. 检查是否足够
            if len(relevant_papers) >= 20:
                break
            
            # 5. 改写查询继续检索
            query = await self.refine_query(query, relevant_papers)
        
        # 6. Cross-Encoder重排序
        final_papers = await self.reranker.rerank(query, relevant_papers, top_k=20)
        
        return final_papers
```
