# src/agents_v2/optimization/dspy_optimizer.py
import dspy
from dspy.teleprompt import BootstrapFewShot, MIPRO

class OutlineOptimizer:
    """大纲生成 Prompt 优化器"""
    
    def __init__(self, llm):
        self.llm = dspy.OpenAI(model=llm)
        dspy.settings.configure(lm=self.llm)
    
    def optimize(self, training_examples):
        """使用 MIPRO 优化 Prompt"""
        # 定义签名
        class GenerateOutline(dspy.Signature):
            """Generate academic paper outline from papers"""
            papers = dspy.InputField(desc="List of relevant papers")
            query = dspy.InputField(desc="User research query")
            outline = dspy.OutputField(desc="Structured outline with sections")
        
        # 创建模块
        outline_module = dspy.ChainOfThought(GenerateOutline)
        
        # 优化
        optimizer = MIPRO(
            metric=self.outline_quality_metric,
            num_candidates=10,
            init_temperature=1.0
        )
        
        optimized = optimizer.compile(
            outline_module,
            trainset=training_examples,
            num_trials=20
        )
        
        return optimized
    
    def outline_quality_metric(self, example, prediction):
        """评估大纲质量"""
        score = 0.0
        outline = prediction.outline
        
        # 检查章节完整性
        required_sections = ["Introduction", "Methods", "Conclusion"]
        for section in required_sections:
            if section.lower() in outline.lower():
                score += 0.3
        
        # 检查引用数量
        if outline.count("[") >= 3:
            score += 0.1
        
        return score
```

# src/agents_v2/optimization/prompt_cache.py
import hashlib
import redis
from typing import Optional

class PromptCache:
    """Prompt 缓存系统"""
    
    def __init__(self, redis_url: str = "redis://localhost:6379"):
        self.redis = redis.from_url(redis_url)
        self.ttl = 3600 * 24 * 7  # 7天
    
    def get(self, prompt: str, model: str) -> Optional[str]:
        """获取缓存的响应"""
        key = self._make_key(prompt, model)
        cached = self.redis.get(key)
        if cached:
            return cached.decode('utf-8')
        return None
    
    def set(self, prompt: str, model: str, response: str):
        """缓存响应"""
        key = self._make_key(prompt, model)
        self.redis.setex(key, self.ttl, response)
    
    def _make_key(self, prompt: str, model: str) -> str:
        """生成缓存键"""
        content = f"{model}:{prompt}"
        return f"prompt_cache:{hashlib.sha256(content.encode()).hexdigest()}"
```
