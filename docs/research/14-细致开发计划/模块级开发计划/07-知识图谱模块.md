## 1. 模块定位

知识图谱处在证据表之后、Scope 和生成模块之前。它的核心价值是把分散的 EvidenceRecord 组织成图结构，使系统能够回答：

```text
哪些论文属于同一主题？
哪些论文使用相似方法？
哪些发现支持同一结论？
哪些局限共同指向同一个研究空白？
某个 gap 背后有哪些论文和证据？
两个主题、方法或论文之间通过什么路径相关？
当前选择的图谱子图能限定哪些 paper_ids 和 evidence_ids？
```

产品主线：

```text
研究项目
  -> 论文库
  -> PDF 解析与分块
  -> 论文卡片
  -> 证据表
  -> 知识图谱          <-- 本模块
  -> Retrieval Scope
  -> Scope-based QA
  -> 文献综述
  -> 创新点报告
```

## 3. 数据提取方案

### 3.1 输入数据结构

当前可直接用于构图的数据源：

```text
EvidenceRecord（13字段）：
  evidence_id, paper_id, topic, research_question, method,
  data_or_sample, finding, limitation, future_work,
  evidence_strength, citation_context, source_chunk_id, source_quote

PaperCard（模型中定义）：
  paper_id, title, authors, year, topics, method,
  key_findings, limitations, possible_gaps, future_work,
  data_or_sample, research_question, abstract...

Paper（模型中定义）：
  paper_id, title, authors, year, doi, venue,
  abstract, references...
```

### 3.2 提取映射表

从现有字段到图谱节点/边的映射：

```python
def _add_task_node(self, nodes, edges, paper_node_id, research_question, ev):
    """从 EvidenceRecord.research_question 构建 Task 节点"""
    if not research_question or research_question.strip().lower() in ("", "unknown", "n/a"):
        return
    for rq in self._split_multi(research_question):
        normalized = normalize_label(rq)
        if not normalized:
            continue
        canonical = resolve_alias(normalized)
        task_id = f"task:{stable_hash(canonical)}"
        existing = next((n for n in nodes if n.node_id == task_id), None)
        if existing:
            self._merge_provenance(existing, ev)
        else:
            nodes.append(GraphNode(
                node_id=task_id,
                node_type=NodeType.TASK,
                label=rq.strip(),
                properties={
                    "normalized_label": canonical,
                    "paper_ids": [ev.paper_id],
                    "evidence_ids": [ev.evidence_id],
                    "source_chunk_ids": [ev.source_chunk_id] if ev.source_chunk_id else [],
                    "source_quotes": [ev.source_quote] if ev.source_quote else [],
                }
            ))
        edge_id = f"edge:STUDIES_TASK:{stable_hash(paper_node_id)}:{stable_hash(task_id)}"
        existing_edge = next((e for e in edges if e.edge_id == edge_id), None)
        if existing_edge:
            self._merge_edge_provenance(existing_edge, ev)
        else:
            edges.append(GraphEdge(
                edge_id=edge_id,
                source_id=paper_node_id,
                target_id=task_id,
                edge_type=EdgeType.STUDIES_TASK,
                properties={
                    "paper_ids": [ev.paper_id],
                    "evidence_ids": [ev.evidence_id],
                    "source_chunk_ids": [ev.source_chunk_id] if ev.source_chunk_id else [],
                }
            ))
```

#### G4.2 resolve_graph_scope 实现

**修改文件**：`src/agents_v3/research_workspace/graph_service.py`

**新增方法**：

```python
def resolve_graph_scope(
    self,
    project_id: str,
    graph_node_ids: list[str],
    graph_edge_ids: list[str] | None = None,
    hops: int = 1,
) -> dict:
    """
    将用户选择的图谱节点/边解析为 paper_ids + evidence_ids。
    供 RetrievalScopeService 使用。
    """
    graph = self.get_graph(project_id)
    if not graph:
        return {"paper_ids": [], "evidence_ids": [], "warnings": ["graph_not_found"]}

    # 从选中节点出发，获取子图
    subgraph = self.get_subgraph(project_id, graph_node_ids, hops=hops)

    # 收集所有 paper_ids 和 evidence_ids
    paper_ids = list(set(subgraph.get("related_paper_ids", [])))
    evidence_ids = list(set(subgraph.get("related_evidence_ids", [])))

    # 如果选中的是 Gap 节点，额外包含 supporting evidence
    for node_id in graph_node_ids:
        node = self.get_node(project_id, node_id)
        if node and node.node_type == NodeType.GAP:
            supporting = node.properties.get("supporting_evidence_ids", [])
            evidence_ids.extend(supporting)
            supporting_papers = node.properties.get("supporting_paper_ids", [])
            paper_ids.extend(supporting_papers)

    paper_ids = list(set(paper_ids))
    evidence_ids = list(set(evidence_ids))

    # 生成 scope summary
    node_map = {n.node_id: n for n in graph.nodes}
    selected_labels = [node_map[nid].label for nid in graph_node_ids if nid in node_map]
    summary = f"图谱子图范围：{', '.join(selected_labels[:5])}，包含 {len(paper_ids)} 篇论文、{len(evidence_ids)} 条证据"

    warnings = []
    if not paper_ids:
        warnings.append("scope_no_papers")
    if not evidence_ids:
        warnings.append("scope_no_evidence")
    if hops > 2:
        warnings.append("hops_truncated")

    return {
        "paper_ids": paper_ids,
        "evidence_ids": evidence_ids,
        "graph_node_ids": subgraph.get("selected_node_ids", graph_node_ids),
        "graph_edge_ids": [e.edge_id for e in subgraph.get("edges", [])],
        "scope_summary": summary,
        "warnings": warnings,
    }
```

**验收**：
- 选择 Topic 节点 → 解析出相关 paper_ids + evidence_ids
- 选择 Gap 节点 → 解析出 supporting_paper_ids + supporting_evidence_ids
- hops=2 时返回 warning
- 空 scope 返回 warnings

```python
def search_nodes(
    self,
    project_id: str,
    query: str,
    node_types: list[str] | None = None,
    top_k: int = 10,
) -> list[dict]:
    """简单 label 包含匹配搜索节点。P1 可接入 embedding/BM25。"""
    graph = self.get_graph(project_id)
    if not graph:
        return []
    query_lower = query.lower().strip()
    results = []
    for node in graph.nodes:
        if node_types and node.node_type.value not in node_types:
            continue
        label = node.label.lower()
        normalized = node.properties.get("normalized_label", "").lower()
        if query_lower in label or query_lower in normalized:
            results.append({
                "node_id": node.node_id,
                "node_type": node.node_type.value,
                "label": node.label,
                "paper_count": len(node.properties.get("paper_ids", [])),
                "evidence_count": len(node.properties.get("evidence_ids", [])),
            })
    return results[:top_k]
```

#### G5.2 build_graph_context 构建 GraphRAG 上下文

```python
def build_graph_context(
    self,
    project_id: str,
    graph_node_ids: list[str],
    hops: int = 1,
    token_budget: int = 4000,
) -> dict:
    """
    为 QA 构建紧凑的 GraphRAG 上下文。
    输出格式：
    [Graph Scope]
    Selected nodes: ...
    Related papers: p1, p2, p3

    [Key relations]
    Paper A --USES_METHOD--> Method X (ev1)
    Paper B --HAS_LIMITATION--> Limitation Y (ev2)

    [Evidence]
    ev1 | Paper A | method | quote...
    """
    scope = self.resolve_graph_scope(project_id, graph_node_ids, hops=hops)
    graph = self.get_graph(project_id)
    node_map = {n.node_id: n for n in graph.nodes}

    # 构建关系描述
    relations = []
    subgraph = self.get_subgraph(project_id, graph_node_ids, hops=hops)
    for edge in subgraph.get("edges", []):
        src = node_map.get(edge.source_id)
        tgt = node_map.get(edge.target_id)
        if src and tgt:
            ev_ids = edge.properties.get("evidence_ids", [])[:2]
            relations.append(
                f"{src.label} --{edge.edge_type.value}--> {tgt.label}"
                + (f" ({', '.join(ev_ids)})" if ev_ids else "")
            )

    # 构建上下文文本
    context_parts = [
        "[Graph Scope]",
        f"Selected: {', '.join(node_map[nid].label for nid in graph_node_ids if nid in node_map)}",
        f"Hops: {hops}",
        f"Related papers: {', '.join(scope['paper_ids'][:10])}",
        f"Related evidence: {len(scope['evidence_ids'])} records",
        "",
        "[Key Relations]",
        *relations[:20],
    ]

    text = "\n".join(context_parts)
    estimated_tokens = len(text) // 2  # 粗估

    return {
        "context_text": text,
        "paper_ids": scope["paper_ids"],
        "evidence_ids": scope["evidence_ids"],
        "estimated_tokens": estimated_tokens,
        "warnings": scope.get("warnings", []),
    }
```

#### G5.3 Topic/Gap 全局摘要

```python
def build_graph_summary(self, project_id: str) -> dict:
    """
    为全局问题生成 Topic/Gap 层面的摘要。
    用于综述生成和全局 QA。
    """
    graph = self.get_graph(project_id)
    if not graph:
        return {}

    # Topic 统计
    topic_nodes = [n for n in graph.nodes if n.node_type == NodeType.TOPIC]
    topic_stats = []
    for t in sorted(topic_nodes, key=lambda n: len(n.properties.get("paper_ids", [])), reverse=True):
        topic_stats.append({
            "label": t.label,
            "paper_count": len(t.properties.get("paper_ids", [])),
            "evidence_count": len(t.properties.get("evidence_ids", [])),
        })

    # Method 统计
    method_nodes = [n for n in graph.nodes if n.node_type == NodeType.METHOD]
    method_stats = []
    for m in sorted(method_nodes, key=lambda n: len(n.properties.get("paper_ids", [])), reverse=True):
        method_stats.append({
            "label": m.label,
            "paper_count": len(m.properties.get("paper_ids", [])),
        })

    # Gap 统计
    gap_nodes = [n for n in graph.nodes if n.node_type == NodeType.GAP]
    gap_stats = []
    for g in sorted(gap_nodes, key=lambda n: n.properties.get("confidence", 0), reverse=True):
        gap_stats.append({
            "label": g.label,
            "confidence": g.properties.get("confidence", 0),
            "supporting_paper_count": len(g.properties.get("supporting_paper_ids", [])),
        })

    return {
        "top_topics": topic_stats[:10],
        "top_methods": method_stats[:10],
        "top_gaps": gap_stats[:10],
        "total_papers": len([n for n in graph.nodes if n.node_type == NodeType.PAPER]),
        "total_evidence": len(set(
            eid for n in graph.nodes
            for eid in n.properties.get("evidence_ids", [])
        )),
    }
```

**验收**：
- 局部问题能从节点子图取证据
- 全局问题能从 Topic/Gap summary 取摘要
- 上下文中每条事实都有 evidence_id
- token_budget 超限时有截断

```python
def _detect_sparse_regions(self, graph: KnowledgeGraph, min_degree: int = 2) -> list[dict]:
    """
    找到 degree < threshold 的节点集合。
    稀疏区域可能表示：新兴主题、边缘方法、或研究空白。
    """
    adj = self._build_adjacency(graph)
    sparse = []
    for node in graph.nodes:
        if node.node_type == NodeType.PAPER:
            continue
        degree = len(adj.get(node.node_id, []))
        if degree < min_degree:
            sparse.append({
                "node_id": node.node_id,
                "node_type": node.node_type.value,
                "label": node.label,
                "degree": degree,
            })
    return sparse
```

#### G6.2 Method-Dataset 矩阵 Gap

```python
def _method_dataset_gaps(self, graph: KnowledgeGraph) -> list[dict]:
    """
    如果某 Method 从未在某 Dataset 上使用，可能是 gap。
    借鉴 Consensus 的 Research Gap Matrix。
    """
    # 构建 Method->Paper 和 Dataset->Paper 映射
    method_papers = {}
    dataset_papers = {}
    for edge in graph.edges:
        if edge.edge_type == EdgeType.USES_METHOD:
            method_papers.setdefault(edge.target_id, set()).add(edge.source_id)
        elif edge.edge_type == EdgeType.USES_DATASET:
            dataset_papers.setdefault(edge.target_id, set()).add(edge.source_id)

    gaps = []
    for method_id, mpapers in method_papers.items():
        for dataset_id, dpapers in dataset_papers.items():
            overlap = mpapers & dpapers
            if not overlap and len(mpapers) >= 2 and len(dpapers) >= 2:
                gaps.append({
                    "method_id": method_id,
                    "dataset_id": dataset_id,
                    "method_paper_count": len(mpapers),
                    "dataset_paper_count": len(dpapers),
                    "type": "method_dataset_gap",
                })
    return gaps
```

#### G6.3 Future Work 交集聚合增强

当前 `_aggregate_gaps` 只按 limitation 聚合。增强为：

```python
def _aggregate_gaps_v2(self, nodes, edges, evidence_records, paper_cards):
    """
    增强版 Gap 聚合：
    1. 从 limitation 聚类（已有）
    2. 从 future_work 聚类（新增）
    3. 从 possible_gaps 聚类（新增）
    4. limitation + future_work 交叉增强（新增）
    """
    # 收集所有候选
    candidates = []

    # 来源1: limitation 节点
    for node in nodes:
        if node.node_type == NodeType.LIMITATION:
            candidates.append({
                "text": node.properties.get("normalized_label", node.label),
                "source_type": "limitation",
                "node_id": node.node_id,
                "paper_ids": node.properties.get("paper_ids", []),
                "evidence_ids": node.properties.get("evidence_ids", []),
            })

    # 来源2: evidence_record.future_work
    for ev in evidence_records:
        if ev.future_work and ev.future_work.strip().lower() not in ("", "unknown", "n/a"):
            normalized = normalize_label(ev.future_work)
            candidates.append({
                "text": normalized,
                "source_type": "future_work",
                "node_id": None,
                "paper_ids": [ev.paper_id],
                "evidence_ids": [ev.evidence_id],
            })

    # 来源3: paper_card.possible_gaps
    for card in paper_cards:
        if hasattr(card, "possible_gaps") and card.possible_gaps:
            for gap_text in card.possible_gaps:
                normalized = normalize_label(gap_text)
                candidates.append({
                    "text": normalized,
                    "source_type": "possible_gap",
                    "node_id": None,
                    "paper_ids": [card.paper_id],
                    "evidence_ids": [],
                })

    # 聚类: 按 normalized text 相似度分组
    # MVP 用精确匹配，P1 用 embedding 相似度
    clusters = {}
    for c in candidates:
        key = c["text"]
        if key not in clusters:
            clusters[key] = []
        clusters[key].append(c)

    # 生成 Gap 节点
    for key, cluster in clusters.items():
        source_types = list(set(c["source_type"] for c in cluster))
        all_paper_ids = list(set(p for c in cluster for p in c["paper_ids"]))
        all_evidence_ids = list(set(e for c in cluster for e in c["evidence_ids"]))

        confidence = compute_gap_confidence(
            source_types=source_types,
            evidence_strength="medium",  # TODO: 从 evidence_records 聚合
            supporting_paper_count=len(all_paper_ids),
            has_quote=bool(all_evidence_ids),
        )

        if confidence < 0.5 and "possible_gap" not in source_types:
            if len(all_paper_ids) < 2:
                continue

        gap_id = f"gap:{stable_hash(key)}"
        # ... 创建 Gap 节点和 SUGGESTS_GAP 边（同现有逻辑）
```

**验收**：
- 单条普通 limitation 不无条件生成 Gap
- 多篇论文共同 limitation 聚合成同一 Gap
- future_work 能生成高置信 candidate
- possible_gaps 能生成 candidate
- Method-Dataset 矩阵空位能识别为潜在 Gap

```python
def _build_ghost_papers(self, nodes, edges, papers):
    """
    GhostPaper: 被库中论文引用但本身不在库中的论文。
    借鉴 Literature Mapper。
    """
    # 收集所有 references
    all_refs = {}  # normalized_title -> cited_by_paper_ids
    paper_ids_in_project = {p.paper_id for p in papers}

    for paper in papers:
        if not hasattr(paper, "references") or not paper.references:
            continue
        for ref in paper.references:
            ref_key = normalize_label(ref.get("title", ""))
            if ref_key:
                all_refs.setdefault(ref_key, {"cited_by": [], "doi": ref.get("doi", "")})
                all_refs[ref_key]["cited_by"].append(paper.paper_id)

    # 找出被引用 >= 2 次且不在库中的
    for ref_key, info in all_refs.items():
        if len(info["cited_by"]) >= 2:
            ghost_id = f"ghost:{stable_hash(ref_key)}"
            # 检查是否已在库中
            if not any(normalize_label(p.title) == ref_key for p in papers if p.title):
                nodes.append(GraphNode(
                    node_id=ghost_id,
                    node_type=NodeType.GHOST_PAPER,
                    label=ref_key,
                    properties={
                        "cited_by_count": len(info["cited_by"]),
                        "cited_by_paper_ids": info["cited_by"],
                        "doi": info["doi"],
                        "source": "reference_extraction",
                    }
                ))
```

```python
def _build_cites_edges(self, nodes, edges, papers):
    """从 Paper.references 构建 CITES 边。"""
    paper_id_set = {p.paper_id for p in papers}
    title_to_paper = {normalize_label(p.title): p.paper_id for p in papers if p.title}

    for paper in papers:
        if not hasattr(paper, "references") or not paper.references:
            continue
        for ref in paper.references:
            ref_title = normalize_label(ref.get("title", ""))
            cited_id = title_to_paper.get(ref_title)
            if cited_id and cited_id in paper_id_set:
                edge_id = f"edge:CITES:{stable_hash(paper.paper_id)}:{stable_hash(cited_id)}"
                edges.append(GraphEdge(
                    edge_id=edge_id,
                    source_id=f"paper:{paper.paper_id}",
                    target_id=f"paper:{cited_id}",
                    edge_type=EdgeType.CITES,
                    properties={"source": "reference_extraction"}
                ))
```

```python
def normalize_label(text: str) -> str:
    if not text:
        return ""
    # 1. strip
    text = text.strip()
    # 2. lowercase
    text = text.lower()
    # 3. 全角转半角
    text = unicodedata.normalize("NFKC", text)
    # 4. 去除多余空白
    text = re.sub(r"\s+", " ", text)
    # 5. 去除末尾标点
    text = text.rstrip(".,;:。，；：")
    # 6. 中文标点归一
    text = text.replace("，", ",").replace("。", ".").replace("；", ";")
    # 7. hyphen/underscore 统一
    text = text.replace("_", " ").replace("-", " ")
    # 8. 常见复数归一（仅英文）
    if text.endswith("es") and len(text) > 4:
        text = text[:-2]
    elif text.endswith("s") and not text.endswith("ss") and len(text) > 3:
        text = text[:-1]
    # 9. alias 替换
    text = resolve_alias(text)
    return text
```
